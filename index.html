// Link CSV untuk Membaca Semua Data const CSV_URL = "https://docs.google.com/spreadsheets/d/e/2PACX-1vRa50Mr58hg80Aj-iuGNIl2PkXgivx9cKcMN1iRgh-2ixbY7raFA1pDBtoUYa9oJs9w9lcHVaEy3yef/pub?gid=498101163&single=true&output=csv"; // Masukkan URL Google Apps Script Web App Hasil Deploy Kamu Di Sini! const WEB_APP_URL = "GANTI_DENGAN_URL_WEB_APP_HASIL_DEPLOY_APPS_SCRIPT"; let picChartInstance = null; let destChartInstance = null; let startTime = null; let timerInterval = null; function switchTab(tabName) { const pageInput = document.getElementById('pageInput'); const pageDashboard = document.getElementById('pageDashboard'); const btnInput = document.getElementById('btnTabInput'); const btnDashboard = document.getElementById('btnTabDashboard'); if (tabName === 'input') { pageInput.style.display = 'block'; pageDashboard.style.display = 'none'; btnInput.classList.add('active'); btnDashboard.classList.remove('active'); } else { pageInput.style.display = 'none'; pageDashboard.style.display = 'block'; btnInput.classList.remove('active'); btnDashboard.classList.add('active'); } } async function loadSpreadsheetData() { try { const response = await fetch(CSV_URL); const csvText = await response.text(); const rows = parseCSV(csvText); const dataList = []; for (let i = 1; i < rows.length; i++) { const r = rows[i]; if (r.length > 1 && r.some(cell => cell.trim() !== '')) { dataList.push({ bulan: r[0] || '-', waktu: r[1] || '-', pic: r[2] || 'Unassigned', tujuan: r[3] || 'Lainnya', jenisMobil: r[4] || '-', box: r[5] || '0', durasi: r[6] || '-', statusTarget: r[7] || '-', fotoMulai: r[8] || '', fotoSelesai: r[9] || '' }); } } updateDashboardUI(dataList); const now = new Date(); document.getElementById('lastUpdate').innerText = `(${now.getHours().toString().padStart(2,'0')}.${now.getMinutes().toString().padStart(2,'0')}.${now.getSeconds().toString().padStart(2,'0')})`; } catch (err) { console.error("Gagal membaca CSV:", err); } } function parseCSV(text) { let lines = text.split('\n'); return lines.map(line => { let values = []; let insideQuote = false; let value = ''; for (let char of line) { if (char === '"' && insideQuote) { insideQuote = false; } else if (char === '"' && !insideQuote) { insideQuote = true; } else if (char === ',' && !insideQuote) { values.push(value.trim()); value = ''; } else { value += char; } } values.push(value.trim()); return values; }); } function updateDashboardUI(data) { const tbody = document.getElementById('dashboardTableBody'); tbody.innerHTML = ''; let achieve = 0; let notAchieve = 0; const picStats = {}; const destStats = {}; data.forEach(item => { const status = (item.statusTarget || '').toLowerCase(); const isAchieve = status.includes('achieve') && !status.includes('not'); if (isAchieve) achieve++; else notAchieve++; if (!picStats[item.pic]) picStats[item.pic] = { achieve: 0, notAchieve: 0 }; if (isAchieve) picStats[item.pic].achieve++; else picStats[item.pic].notAchieve++; if (item.tujuan) { destStats[item.tujuan] = (destStats[item.tujuan] || 0) + 1; } const tr = document.createElement('tr'); tr.innerHTML = ` ${item.bulan} ${item.waktu} ${item.pic} ${item.tujuan} ${item.jenisMobil} ${item.box} ${item.durasi} ${item.statusTarget} ${item.fotoMulai ? `` : '-'} ${item.fotoSelesai ? `` : '-'} `; tbody.appendChild(tr); }); const total = achieve + notAchieve; const rate = total > 0 ? ((achieve / total) * 100).toFixed(1) : 0; document.getElementById('kpiTotal').innerText = total; document.getElementById('kpiAchieve').innerText = achieve; document.getElementById('kpiNotAchieve').innerText = notAchieve; document.getElementById('kpiRate').innerText = `${rate}%`; renderPICChart(picStats); renderDestinationChart(destStats); } function renderPICChart(picStats) { const labels = Object.keys(picStats); const achieveData = labels.map(p => picStats[p].achieve); const notAchieveData = labels.map(p => picStats[p].notAchieve); const ctx = document.getElementById('picChart').getContext('2d'); if (picChartInstance) picChartInstance.destroy(); picChartInstance = new Chart(ctx, { type: 'bar', data: { labels: labels, datasets: [ { label: 'Achieve', data: achieveData, backgroundColor: '#22c55e' }, { label: 'Not Achieve', data: notAchieveData, backgroundColor: '#ef4444' } ] }, options: { responsive: true, maintainAspectRatio: false, scales: { x: { stacked: true }, y: { stacked: true, beginAtZero: true } }, plugins: { legend: { position: 'top' } } } }); } function renderDestinationChart(destStats) { const labels = Object.keys(destStats); const data = Object.values(destStats); const ctx = document.getElementById('destinationChart').getContext('2d'); if (destChartInstance) destChartInstance.destroy(); destChartInstance = new Chart(ctx, { type: 'doughnut', data: { labels: labels, datasets: [{ data: data, backgroundColor: [ '#3b82f6', '#10b981', '#f59e0b', '#8b5cf6', '#ec4899', '#6366f1', '#14b8a6', '#f97316' ] }] }, options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { position: 'right' } } } }); } // --- FUNGSI TIMER & SUBMIT BARU (SIMPAN KE GOOGLE SHEETS) --- function startLoading() { startTime = new Date(); document.getElementById('timerDisplay').innerText = "00:00:00"; if (timerInterval) clearInterval(timerInterval); timerInterval = setInterval(() => { const diff = new Date() - startTime; const hrs = Math.floor(diff / 3600000).toString().padStart(2, '0'); const mins = Math.floor((diff % 3600000) / 60000).toString().padStart(2, '0'); const secs = Math.floor((diff % 60000) / 1000).toString().padStart(2, '0'); document.getElementById('timerDisplay').innerText = `${hrs}:${mins}:${secs}`; }, 1000); alert('Proses loading dimulai!'); } async function stopAndSubmit() { if (!startTime) { alert('Klik "Mulai Loading" terlebih dahulu!'); return; } clearInterval(timerInterval); const now = new Date(); const namaBulan = ["Januari", "Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "November", "Desember"]; const payload = { bulan: namaBulan[now.getMonth()], waktu: `${now.getDate()}/${now.getMonth() + 1}/${now.getFullYear()}`, pic: document.getElementById('pic').value, tujuan: document.getElementById('tujuan').value, jenisMobil: document.getElementById('jenisMobil').value, jumlahBox: document.getElementById('jumlahBox').value, durasi: document.getElementById('timerDisplay').innerText, statusTarget: document.getElementById('statusTarget').value, fotoMulai: '', // Opsional, tambahkan variabel base64 jika menggunakan foto fotoSelesai: '' }; try { // Mengirim data ke Google Spreadsheet await fetch(WEB_APP_URL, { method: 'POST', mode: 'no-cors', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); alert('Data berhasil disimpan ke Spreadsheet!'); document.getElementById('shipmentForm').reset(); // Beri jeda 2 detik agar Google Sheets selesai menulis baris baru sebelum reload setTimeout(async () => { await loadSpreadsheetData(); switchTab('dashboard'); }, 2000); } catch (error) { console.error("Gagal mengirim data:", error); alert("Gagal mengirim data ke Spreadsheet."); } } window.onload = loadSpreadsheetData;
