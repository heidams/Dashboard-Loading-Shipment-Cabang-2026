// URL CSV Publish dari Google Sheets kamu const SCRIPT_URL = "https://docs.google.com/spreadsheets/d/e/2PACX-1vRa50Mr58hg80Aj-iuGNIl2PkXgivx9cKcMN1iRgh-2ixbY7raFA1pDBtoUYa9oJs9w9lcHVaEy3yef/pub?gid=498101163&single=true&output=csv"; let myChart = null; let startTime = null; let timerInterval = null; // --- 1. LOGIKA TAB --- function switchTab(tabName) { const pageInput = document.getElementById('pageInput'); const pageDashboard = document.getElementById('pageDashboard'); const btnInput = document.getElementById('btnTabInput'); const btnDashboard = document.getElementById('btnTabDashboard'); if (tabName === 'input') { pageInput.classList.add('active'); pageDashboard.classList.remove('active'); btnInput.classList.add('active'); btnDashboard.classList.remove('active'); } else { pageInput.classList.remove('active'); pageDashboard.classList.add('active'); btnInput.classList.remove('active'); btnDashboard.classList.add('active'); } } // --- 2. BACA & PARSE DATA CSV DARI SPREADSHEET --- async function loadDataFromSpreadsheet() { try { const response = await fetch(SCRIPT_URL); const csvText = await response.text(); const rows = parseCSV(csvText); // Abaikan baris pertama (Header) const dataList = []; for (let i = 1; i < rows.length; i++) { const row = rows[i]; if (row.length > 1) { dataList.push({ waktu: row[0] || '-', pic: row[1] || '-', tujuan: row[2] || '-', jenisMobil: row[3] || '-', jumlahBox: row[4] || '0', statusTarget: row[5] || '-', durasi: row[6] || '-', fotoMulai: row[7] || '', fotoSelesai: row[8] || '' }); } } renderDashboard(dataList); } catch (error) { console.error("Gagal mengambil data CSV:", error); } } // Helper untuk parsing format CSV dengan aman function parseCSV(text) { let lines = text.split('\n'); return lines.map(line => { let values = []; let insideQuote = false; let value = ''; for (let char of line) { if (char === '"' && insideQuote) { insideQuote = false; } else if (char === '"' && !insideQuote) { insideQuote = true; } else if (char === ',' && !insideQuote) { values.push(value.trim()); value = ''; } else { value += char; } } values.push(value.trim()); return values; }); } function renderDashboard(dataList) { const tbody = document.getElementById('dashboardTableBody'); tbody.innerHTML = ''; let achieveCount = 0; let notAchieveCount = 0; if (dataList.length === 0) { tbody.innerHTML = 'Belum ada data...'; return; } dataList.forEach(item => { if (item.statusTarget === 'Achieve') achieveCount++; else if (item.statusTarget === 'Not Achieve') notAchieveCount++; const tr = document.createElement('tr'); tr.innerHTML = ` ${item.waktu} ${item.pic} ${item.tujuan} ${item.jenisMobil} ${item.jumlahBox} ${item.durasi} ${item.statusTarget} ${item.fotoMulai ? `` : '-'} ${item.fotoSelesai ? `` : '-'} `; tbody.appendChild(tr); }); updateChart(achieveCount, notAchieveCount); } // --- 3. KOMPRESI GAMBAR --- function compressImage(file, maxWidth = 1024, maxHeight = 1024, quality = 0.7) { return new Promise((resolve, reject) => { if (!file) { resolve(""); return; } const reader = new FileReader(); reader.readAsDataURL(file); reader.onload = (event) => { const img = new Image(); img.src = event.target.result; img.onload = () => { let width = img.width; let height = img.height; if (width > height) { if (width > maxWidth) { height = Math.round((height * maxWidth) / width); width = maxWidth; } } else { if (height > maxHeight) { width = Math.round((width * maxHeight) / height); height = maxHeight; } } const canvas = document.createElement('canvas'); canvas.width = width; canvas.height = height; const ctx = canvas.getContext('2d'); ctx.drawImage(img, 0, 0, width, height); resolve(canvas.toDataURL('image/jpeg', quality)); }; img.onerror = (err) => reject(err); }; reader.onerror = (err) => reject(err); }); } // --- 4. TIMER & SUBMIT DUMMY/LOCAL --- function startLoading() { startTime = new Date(); document.getElementById('timerDisplay').innerText = "00:00:00"; if (timerInterval) clearInterval(timerInterval); timerInterval = setInterval(() => { const diff = new Date() - startTime; const hrs = Math.floor(diff / 3600000).toString().padStart(2, '0'); const mins = Math.floor((diff % 3600000) / 60000).toString().padStart(2, '0'); const secs = Math.floor((diff % 60000) / 1000).toString().padStart(2, '0'); document.getElementById('timerDisplay').innerText = `${hrs}:${mins}:${secs}`; }, 1000); alert('Proses loading dimulai!'); } async function stopAndSubmit() { if (!startTime) { alert('Klik "Mulai Loading" terlebih dahulu!'); return; } clearInterval(timerInterval); const fileMulai = document.getElementById('fotoMulai').files[0]; const fileSelesai = document.getElementById('fotoSelesai').files[0]; const imgMulaiBase64 = await compressImage(fileMulai); const imgSelesaiBase64 = await compressImage(fileSelesai); alert('Data terkompresi berhasil diproses! Silakan perbarui Spreadsheet untuk melihat sinkronisasi.'); // Muat ulang data dari CSV await loadDataFromSpreadsheet(); switchTab('dashboard'); } // --- 5. INITIALIZE CHART --- function initChart() { const ctx = document.getElementById('shipmentChart').getContext('2d'); myChart = new Chart(ctx, { type: 'bar', data: { labels: ['Achieve', 'Not Achieve'], datasets: [{ label: 'Total Shipment', data: [0, 0], backgroundColor: ['#22c55e', '#ef4444'] }] }, options: { responsive: true, maintainAspectRatio: false, plugins: { title: { display: true, text: 'Grafik Status Target Shipment' } } } }); } function updateChart(achieve, notAchieve) { if (myChart) { myChart.data.datasets[0].data = [achieve, notAchieve]; myChart.update(); } } // Jalankan otomatis saat web dibuka window.onload = function() { initChart(); loadDataFromSpreadsheet(); };
